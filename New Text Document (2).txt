// Load all doctors from MySQL (Flask)
function loadDoctors() {
    fetch("http://127.0.0.1:5000/doctors")
    .then(res => res.json())
    .then(data => {
        const tbody = document.getElementById("doctorBody");
        tbody.innerHTML = "";

        data.forEach(doc => {
            let row = tbody.insertRow();
            row.onclick = () => openDoctor(doc);
            row.innerHTML = `
    <td>${doc.doc_id}</td>
    <td>${doc.doctor_name}</td>
    <td>${doc.specialization}</td>
    <td>${doc.department}</td>
    <td>${doc.available_today == 1 ? "Yes" : "No"}</td>
    <td><button onclick="deleteDoctor(${doc.doc_id})">Delete</button></td>
`;
        });
    });
}
function showDoctors(){

fetch("http://127.0.0.1:5000/doctors")
.then(res=>res.json())
.then(data=>{

let html="<table>";

data.forEach(d=>{
html+=`<tr>
<td>${d.doc_id}</td>
<td>${d.doctor_name}</td>
<td>${d.specialization}</td>
</tr>`;
});

html+="</table>";

document.getElementById("queryResult").innerHTML=html;

});

}
// Search doctor
function searchDoctor() {
    let name = document.getElementById("searchDoctor").value;

    fetch(`http://127.0.0.1:5000/search?name=${name}`)
    .then(res => res.json())
    .then(data => {
        const tbody = document.getElementById("doctorBody");
        tbody.innerHTML = "";

        data.forEach(doc => {
            let row = tbody.insertRow();
            row.innerHTML = `
                <td>${doc.doc_id}</td>
                <td>${doc.doctor_name}</td>
                <td>${doc.specialization}</td>
                <td>${doc.department}</td>
                <td>${doc.available_today == 1 ? "Yes" : "No"}</td>
            `;
        });
    });
}

// Available today
function showAvailable() {
    fetch("http://127.0.0.1:5000/available")
    .then(res => res.json())

        tbody.innerHTML = "";

        data.forEach(doc => {

                <td>${doc.specialization}</td>
                <td>${doc.department}</td>
                <td>${doc.available_today == 1 ? "Yes" : "No"}</td>
            `;
        });

}

// ADD doctor to MySQL
function addDoctor() {

    const doctor = {
        doctor_name: document.getElementById("name").value,
        specialization: document.getElementById("spec").value,
        department: document.getElementById("dept").value,
        available_today: document.getElementById("avail").checked ? 1 : 0
    };

    fetch("http://127.0.0.1:5000/add", {
        method: "POST",
        headers: {
            "Content-Type": "application/json"
        },
        body: JSON.stringify(doctor)
    })
    .then(res => res.json())
    .then(data => {
        alert(data.message);
        loadDoctors();
    });
}
function deleteDoctor(id) {

    if (!confirm("Are you sure you want to delete this doctor?"))
        return;

    fetch(`http://127.0.0.1:5000/delete/${id}`, {
        method: "DELETE"
    })
    .then(res => res.json())
    .then(data => {
        alert(data.message);
        loadDoctors();
    });
}
let currentDoctor = null;


function openDoctor(doc){

    currentDoctor = doc;

    document.getElementById("doctorExtra").style.display = "block";
    document.getElementById("selectedDoctorTitle").innerText =
        "Details of " + doc.doctor_name;

    // Load patients
    fetch(`http://127.0.0.1:5000/patients/${doc.doc_id}`)
    .then(res => res.json())
    .then(data => {

        const list = document.getElementById("patientList");
        list.innerHTML = "";

        if(data.length === 0){
            list.innerHTML = "<tr><td colspan='3'>No patients assigned</td></tr>";
        }
        else{
            data.forEach(p => {
                list.innerHTML += `
<tr onclick="togglePatient(${p.patient_id})">
<td>P${p.patient_id}</td>
<td>${p.patient_name}</td>
<td>${p.surgery_done ? "Yes" : "No"}</td>
<td>
<button onclick="event.stopPropagation(); deletePatient(${p.patient_id})">
Delete
</button>
</td>
</tr>

<tr id="patientDetail${p.patient_id}" style="display:none;">
<td colspan="4">

<div class="patientCard">

<h3>Patient: ${p.patient_name}</h3>
<p><b>Doctor:</b> ${currentDoctor.doctor_name}</p>

<h4>Diseases</h4>
<ul id="diseaseList${p.patient_id}"></ul>
<input id="disease${p.patient_id}" placeholder="Add disease">
<button onclick="addDisease(${p.patient_id})">Add</button>

<h4>Symptoms</h4>
<ul id="symptomList${p.patient_id}"></ul>
<input id="symptom${p.patient_id}" placeholder="Add symptom">
<button onclick="addSymptom(${p.patient_id})">Add</button>

<h4>Prescription</h4>
<ul id="drugList${p.patient_id}"></ul>
<input id="drug${p.patient_id}" placeholder="Add drug">
<button onclick="addDrug(${p.patient_id})">Add</button>

</div>

</td>
</tr>
`;
            });
        }
    });

    loadComments(doc.doc_id);
}


function loadComments(id){
    fetch(`http://127.0.0.1:5000/comments/${id}`)
    .then(res=>res.json())
    .then(data=>{
        let box=document.getElementById("commentList");
        box.innerHTML="";

        if(data.length==0)
            box.innerHTML="<p>No comments yet</p>";
        else
            data.forEach(c=>{
                box.innerHTML += `
<p>
• ${c.comment_text}
<button onclick="deleteComment(${c.comment_id})">Delete</button>
</p>
`;
            });
    });
}

function postComment(){
    let text=document.getElementById("newComment").value;

    fetch(`http://127.0.0.1:5000/comments/${currentDoctor.doc_id}`,{
        method:"POST",
        headers:{"Content-Type":"application/json"},
        body:JSON.stringify({text:text})
    })
    .then(res=>res.json())
    .then(()=>{
        document.getElementById("newComment").value="";
        loadComments(currentDoctor.doc_id);
    });
}
function addPatient(){

    let name = document.getElementById("patientName").value;
let disease = document.getElementById("patientDisease").value;
let prescription = document.getElementById("patientPrescription").value;
let surgery = document.getElementById("patientSurgery").checked ? 1 : 0;

    fetch(`http://127.0.0.1:5000/patients/${currentDoctor.doc_id}`,{
        method:"POST",
        headers:{"Content-Type":"application/json"},
        body: JSON.stringify({
name:name,
disease:disease,
prescription:prescription,
surgery_done:surgery
})
    })
    .then(res=>res.json())
    .then(data=>{
        alert(data.message);

        // reload patients list
        openDoctor(currentDoctor);

        document.getElementById("patientName").value="";
        document.getElementById("patientDisease").value="";
    });
}
function deletePatient(id){

    if(!confirm("Delete this patient?"))
        return;

    fetch(`http://127.0.0.1:5000/delete_patient/${id}`,{
        method:"DELETE"
    })
    .then(res=>res.json())
    .then(data=>{
        alert(data.message);
        openDoctor(currentDoctor);
    });
}
function deleteComment(id){

    if(!confirm("Delete this comment?"))
        return;

    fetch(`http://127.0.0.1:5000/delete_comment/${id}`,{
        method:"DELETE"
    })
    .then(res=>res.json())
    .then(()=>{
        loadComments(currentDoctor.doc_id);
    });
}

function togglePatient(id){

let row=document.getElementById("patientDetail"+id);

if(row.style.display==="none"){
row.style.display="table-row";
loadPatientDetails(id);
}
else{
row.style.display="none";
}

}

function toggleVisit(i){

let box=document.getElementById("visit"+i);

if(box.style.display==="none")
box.style.display="block";
else
box.style.display="none";

}

function loadPatientDetails(pid){

});
}
